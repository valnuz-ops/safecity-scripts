/**
 * SafeCity — щотижнева автоперевірка фільтрів у всіх категоріях меню.
 * Файл додається в проєкт «уведомление клиенту» (orders): використовує tg_() з orders.gs
 * і TG_TOKEN / TG_CHAT_ID з config.gs.
 *
 * Що перевіряє на кожній сторінці категорії (як бачить покупець):
 *  1. Дублі значень — різне написання одного значення («RS485» / «RS-485», різні лапки, регістр, пробіли).
 *  2. Склейки — значення через кому («LED, ІЧ») у фільтрі, де покупець обирає одне.
 *  3. Числовий повзунок починається з 0 — у частини товарів поле не заповнене.
 *  4. Фільтр з одним значенням — нічого не відсікає (крім перемикачів «Так»).
 *  5. Фільтр заповнений менш ніж у 30% товарів категорії.
 *  6. Сторінка категорії не відкрилась.
 *
 * У Telegram приходить звіт: спочатку НОВІ проблеми (яких не було минулого тижня), потім підсумок.
 *
 * Запуск:
 *  1. Вставити цей файл у проєкт → Зберегти.
 *  2. Вибрати функцію filtersCheck → Виконати (перший раз — дати дозволи). Має прийти звіт у Telegram.
 *  3. Вибрати installFiltersCheckTrigger → Виконати — далі звіт щопонеділка о 9:00 за Києвом.
 */

const FC_SITE = 'https://safecity.in.ua';
const FC_WEEKDAY = ScriptApp.WeekDay.MONDAY;
const FC_HOUR = 9;
const FC_MIN_FILL = 0.3;            // фільтр заповнений менш ніж у 30% товарів — повідомити
const FC_IGNORE_FILTERS = ['Ціна, грн', 'Наявність'];
// Склейки, які власник прийняв як нормальні (значення описує кілька речей одночасно)
const FC_ALLOWED_COMMA = [
  'DC 5.5/2.1, USB-C',
  'DC 5.5/2.1, RJ45',
  'Від короткого замикання, перевантаження',
  'Датчик диму, тепла та чадного газу',
  'Так, регульований кут фіксації'
];
// Пари, які автоматично схожі, але це різні значення
const FC_NOT_DUPLICATES = [
  ['2.5 мм²', '25 мм²'],
  ['802.3af/at', '802.3af/at/bt']
];

function filtersCheck() {
  const cats = fcCategories_();
  const issues = [];
  let checked = 0;

  const failed = [];
  const started = Date.now();

  // по одній сторінці з паузою — сайт обмежує часті запити і може обірвати з'єднання
  for (let i = 0; i < cats.length; i++) {
    if (Date.now() - started > 5 * 60 * 1000) {        // ліміт Apps Script — 6 хв
      cats.slice(i).forEach(u => failed.push(u + ' (не встигли)'));
      break;
    }
    const url = cats[i];
    const html = fcGet_(FC_SITE + url);
    if (!html) { failed.push(url); continue; }
    const page = fcParse_(html);
    if (!page.sections.length) continue;   // батьківські розділи-вітрини без фільтрів
    checked++;
    fcCheckPage_(url, page).forEach(x => issues.push(x));
    Utilities.sleep(800);
  }

  fcReport_(issues, cats.length, checked, failed);
}

// Завантаження сторінки з повторами: «Адрес недоступен», обрив з'єднання, «Requests Throttled»
function fcGet_(url) {
  for (let attempt = 1; attempt <= 3; attempt++) {
    try {
      const r = UrlFetchApp.fetch(url, { muteHttpExceptions: true, followRedirects: true });
      const html = r.getResponseCode() === 200 ? r.getContentText() : '';
      if (html && !/Requests Throttled/i.test(html.slice(0, 2000))) return html;
      Logger.log(url + ' — спроба ' + attempt + ': код ' + r.getResponseCode());
    } catch (e) {
      Logger.log(url + ' — спроба ' + attempt + ': ' + e.message);
    }
    Utilities.sleep(attempt * 5000);
  }
  return '';
}

// ---------- Категорії з меню сайту ----------
function fcCategories_() {
  const html = fcGet_(FC_SITE + '/');
  if (!html) throw new Error('Головна сторінка сайту не відкрилась (3 спроби)');
  const a = html.indexOf('productsMenu'), b = html.lastIndexOf('productsMenu');
  if (a < 0) throw new Error('Меню каталогу не знайдено на головній — можливо, сайт віддав сторінку-перевірку');
  const end = html.indexOf('</nav>', b) > 0 ? html.indexOf('</nav>', b) : b + 5000;
  const part = html.slice(a, end);
  const set = {};
  (part.match(/href="(\/[^"#?]+\/)"/g) || []).forEach(m => {
    const u = m.slice(6, -1);
    if (!/\/filter\/|\/blog\//.test(u)) set[u] = 1;
  });
  return Object.keys(set);
}

// ---------- Розбір фільтрів сторінки ----------
function fcParse_(html) {
  const sections = [];
  const h1 = (html.match(/<h1[^>]*>([\s\S]*?)<\/h1>/) || [])[1];
  const title = h1 ? fcText_(h1) : '';

  // числові повзунки: назва з data-filer-name форми, межі з init_number_filter(uid, min, max, ...)
  const numRe = /data-filer-name="([^"]+)"[\s\S]*?filter-section-h">([\s\S]*?)<\/div>[\s\S]*?init_number_filter\(\s*"[^"]+",\s*([\d.]+),\s*([\d.]+)/g;
  let m;
  while ((m = numRe.exec(html))) {
    sections.push({ type: 'num', key: m[1], name: fcText_(m[2]), min: +m[3], max: +m[4] });
  }

  // списки значень з лічильниками
  const parts = html.split('j-filter-section').slice(1);
  parts.forEach(p => {
    const nm = p.match(/j-filter-collapse-trigger[^>]*>([\s\S]*?)<\/button>/);
    if (!nm) return;
    const name = fcText_(nm[1]);
    const values = [];
    const vr = /filter-title j-filter-title">([\s\S]*?)<\/span>[\s\S]*?filter-count">(\d+)</g;
    let v;
    const chunk = p.split(/class="filter-section[ "]/)[0];
    while ((v = vr.exec(chunk))) values.push({ label: fcText_(v[1]), count: +v[2] });
    if (values.length) sections.push({ type: 'list', name: name, values: values });
  });

  // друга верстка фільтрів (шаблони Ajax / «Альтернативна енергетика»): filter-group + filter-item
  html.split('j-filter-group"').slice(1).forEach(p => {
    const nm = p.match(/filter-group__title-text">([\s\S]*?)<\/span>/);
    if (!nm) return;
    const name = fcText_(nm[1]);
    if (sections.some(s => s.name === name)) return;
    const values = [];
    const vr = /filter-item__title j-filter-title">([\s\S]*?)<\/span>[\s\S]*?filter-item__quantity">(\d+)</g;
    let v;
    const chunk = p.split('j-filter-group"')[0];
    while ((v = vr.exec(chunk))) values.push({ label: fcText_(v[1]), count: +v[2] });
    if (values.length) sections.push({ type: 'list', name: name, values: values });
  });

  const pres = sections.find(s => s.name === 'Наявність');
  const total = pres ? pres.values.reduce((a, x) => a + x.count, 0) : 0;
  return { title: title, total: total, sections: sections };
}

// ---------- Перевірки ----------
function fcCheckPage_(url, page) {
  const out = [];
  const cat = page.title || url;
  page.sections.forEach(s => {
    if (FC_IGNORE_FILTERS.indexOf(s.name) >= 0) return;

    if (s.type === 'num') {
      if (s.min === 0 && s.max > 0) out.push(fcIssue_(cat, s.name, 'повзунок від 0 — у частини товарів значення не заповнене', url));
      return;
    }

    const vals = s.values;
    // дублі
    const seen = {};
    vals.forEach(v => {
      const k = fcNorm_(v.label);
      if (seen[k] && seen[k] !== v.label && !fcNotDup_(seen[k], v.label)) {
        out.push(fcIssue_(cat, s.name, 'дубль: «' + seen[k] + '» і «' + v.label + '»', url));
      } else seen[k] = v.label;
    });
    // склейки: значення через кому, частина якого — окреме значення цього ж фільтра («LED, ІЧ» при наявних «LED» і «ІЧ»)
    const labels = {};
    vals.forEach(v => labels[v.label.toLowerCase()] = 1);
    vals.forEach(v => {
      if (!/, /.test(v.label) || FC_ALLOWED_COMMA.indexOf(v.label) >= 0 || s.name === 'Бренд') return;
      if (v.label.split(/,\s*/).some(x => labels[x.toLowerCase()])) {
        out.push(fcIssue_(cat, s.name, 'склейка через кому: «' + v.label + '» (' + v.count + ')', url));
      }
    });
    // одне значення в усіх (≥90%) товарів — фільтр нічого не відсікає.
    // Якщо значення лише в частини товарів (напр. «АВР» у 7 з 28 генераторів) — це корисний перемикач, не помилка.
    // Бренд з одним брендом не рахуємо — його Хорошоп показує в усіх розділах.
    if (vals.length === 1 && vals[0].label !== 'Так' && s.name !== 'Бренд' && page.total > 1 &&
        vals[0].count >= page.total * 0.9) {
      out.push(fcIssue_(cat, s.name, 'лише одне значення «' + vals[0].label + '» — фільтр нічого не відсікає', url));
    }
    // мала заповненість
    const filled = vals.reduce((a, x) => a + x.count, 0);
    if (page.total >= 5 && vals.length > 1 && filled < page.total * FC_MIN_FILL) {
      out.push(fcIssue_(cat, s.name, 'заповнено ~' + Math.round(filled / page.total * 100) + '% товарів (' + filled + ' з ' + page.total + ')', url));
    }
  });
  return out;
}

function fcNorm_(s) {
  return s.toLowerCase()
    .replace(/[″”“"'’`ʼ]/g, '')
    .replace(/[\s\-_.,()\/]+/g, '')
    .replace(/х/g, 'x').replace(/с/g, 'c').replace(/а/g, 'a').replace(/о/g, 'o').replace(/р/g, 'p').replace(/е/g, 'e').replace(/і/g, 'i');
}

function fcNotDup_(a, b) {
  return FC_NOT_DUPLICATES.some(p => (p[0] === a && p[1] === b) || (p[0] === b && p[1] === a));
}

function fcIssue_(cat, filter, text, url) {
  return { cat: cat, filter: filter, text: text, url: url || cat, key: cat + '|' + filter + '|' + text.replace(/\(\d+\)|~\d+%|\d+ з \d+/g, '') };
}

function fcText_(s) {
  return s.replace(/<[^>]+>/g, '').replace(/&quot;/g, '"').replace(/&#039;|&apos;/g, "'").replace(/&amp;/g, '&')
    .replace(/&nbsp;/g, ' ').replace(/\s+/g, ' ').trim();
}

// ---------- Звіт ----------
function fcReport_(issues, cats, checked, failed) {
  failed = failed || [];
  const props = PropertiesService.getScriptProperties();
  let prev = {};
  try { prev = JSON.parse(props.getProperty('FC_PREV') || '{}'); } catch (e) {}
  const now = {};
  issues.forEach(x => now[x.key] = x.url);
  // проблеми сторінок, які цього разу не відкрились, не вважаємо виправленими
  Object.keys(prev).forEach(k => { if (failed.some(u => u.split(' ')[0] === prev[k])) now[k] = prev[k]; });
  const fresh = issues.filter(x => !prev[x.key]);
  const gone = Object.keys(prev).filter(k => !now[k]).length;
  try { props.setProperty('FC_PREV', JSON.stringify(now)); } catch (e) { Logger.log('Не вдалося зберегти стан: ' + e); }

  const byType = {};
  issues.forEach(x => {
    const t = x.text.split(/[:—(«]/)[0].trim();
    byType[t] = (byType[t] || 0) + 1;
  });

  const lines = [];
  lines.push('🔎 Перевірка фільтрів SafeCity');
  lines.push('Категорій у меню: ' + cats + ', перевірено з фільтрами: ' + checked);
  if (failed.length) lines.push('⚠️ Не відкрились (' + failed.length + '): ' + failed.slice(0, 15).join(', ') + (failed.length > 15 ? '…' : ''));
  lines.push('Проблем усього: ' + issues.length + (Object.keys(prev).length ? ' (нових ' + fresh.length + ', виправлено ' + gone + ')' : ''));
  Object.keys(byType).sort((a, b) => byType[b] - byType[a]).forEach(t => lines.push('• ' + t + ': ' + byType[t]));
  lines.push('');

  const list = Object.keys(prev).length ? fresh : issues;
  if (list.length) {
    lines.push(Object.keys(prev).length ? '🆕 Нові проблеми:' : 'Перший запуск — усі проблеми:');
    let lastCat = '';
    list.forEach(x => {
      if (x.cat !== lastCat) { lines.push(''); lines.push('▪️ ' + x.cat + ' ' + FC_SITE + x.url); lastCat = x.cat; }
      lines.push('  ' + x.filter + ': ' + x.text);
    });
  } else {
    lines.push('✅ Нових проблем немає.');
  }

  // Telegram — до 4096 символів у повідомленні
  let buf = '';
  lines.forEach(l => {
    if ((buf + '\n' + l).length > 3800) { tg_(buf, false); buf = ''; }
    buf += (buf ? '\n' : '') + l;
  });
  if (buf) tg_(buf, false);
}

// ---------- Тригер ----------
function installFiltersCheckTrigger() {
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'filtersCheck')
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('filtersCheck').timeBased().onWeekDay(FC_WEEKDAY).atHour(FC_HOUR).inTimezone('Europe/Kiev').create();
  Logger.log('Тригер створено: щопонеділка о ' + FC_HOUR + ':00 (Київ)');
}
